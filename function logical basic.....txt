 9..  Write a function to return the factorial of a number.

    cout<<"The factorial of the number"<<ans<<endl;

11...Write a function to swap two numbers (using call by value).

void swap(int a,int b);

int main(){
    int a,b;
    cout<<"enter the number";
    cin>>a>>b;
    cout<<"before"<<"a= "<<a<<" b="<<b<<endl;
    swap(a,b);
    cout<<"after"<<"a= "<<a<<" b="<<b<<endl;
    
    
    return 0;
}
void swap(int a,int b){
    int temp;
    temp=a;
    a=b;
    b=temp;
    cout<<"inner function"<<"a="<<a<<" b="<<b<<endl;
    
    
    
}

 18...Write a function to check whether a number is prime.

#include<iostream>
using namespace std;
void prime(int n);

int main(){
    int n;
    cout<<"Enter the number :";
    cin>>n;
    prime(n);
    
    return 0;
}

void prime(int n){
    int count=0;
    for(int i=1;i<=n;i++){
        if(n%i==0){
        count++;
    }
    }
    if(count==2){
        cout<<"it is a prime number";
    }
    else{
        cout<<"it is not a prime number";
    }
    
    

}
