Question 30: Find Sum of Even and Odd Numbers Separately (1–N)**

Write a C program using a while loop to find the sum of even and odd numbers separately from 1 to N.


#include<iostream>
using namespace std;
int main(){
    int n,i=1,sumeven=0,sumodd=0;
    cout<<"enter the number";
    cin>>n;
    while(i<=n){
        if(i%2==0){
            sumeven=sumeven+i;
        }
        else{
            sumodd=sumodd+i;
        }
        
        i++;
    }
    cout<<"sum of even number ="<<sumeven<<endl;
    cout<<"sum of odd number ="<<sumodd<<endl;
    return 0;
}
