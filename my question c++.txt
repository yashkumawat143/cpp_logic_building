1.. **Question:**

Write a C program that takes a single character as input from the user and performs the following actions based on the entered character:

* If the user enters **'a'**, print numbers from **1 to 10**.
* If the user enters **'b'**, print numbers from **10 to 1**.
* If the user enters **'c'**, , then print only the **even numbers** from 1 to 10.
* If the user enters **'d'**, , then print only the **odd numbers** from 1 to 10.
* If the input does not match any of the above characters, display **"Invalid input"**.




#include<iostream>
using namespace std;
int main(){
    char choice;
    
    cout<<"enter the value : ";
    cin>>choice;
    if(choice=='a'){
        int i,n=10;
        i=1;
        while(i<=n){
            cout<<"Number from 1 to 10 :"<<i<<endl;
            i++;
        }
        
    }
    
    else if(choice=='b'){
        int i=10;
        while(i>=1){
            cout<<"Number from 10 to 1 :"<<i<<endl;
            i--;
        }
    }
    
    else if(choice=='c'){
        int i=1;
        while(i<=10){
            if(i%2==0){
                cout<<i<<endl;
            }
            i++;
        }
        
    }
    else if(choice=='d'){
        int i=10;
        while(i>=1){
            if(i%2!=0){
                cout<<i<<endl;
            }
            i--;
        }
    }
    
    else{
        cout<<"invalid input";
    }
    
    return 0;
}

3.. **Question:**

Write a C program that takes a single character as input and performs different operations based on the input character:

* If the user enters **'s'**, ask the user to input **n numbers** and then print the **sum** of those numbers.
* If the user enters **'a'**, ask the user to input a **number** and check whether it is an **Armstrong number** or not.
* If the user enters **'p'**, ask the user to input a **number** and check whether it is a **Palindrome** or not.
* If the input character is anything else, print **"Invalid input"**.

#include<iostream>
using namespace std;
int main(){
    char choice;
    cout<<"Enter the value :";
    cin>>choice;
    
    if(choice=='s'){
        int n,i=1,sum=0,num;
        cout<<"Enter the total number :";
        cin>>n;
        while(i<=n){
            cout<<"enter the numbers :";
            cin>>num;
            sum=sum+num;
            i++;
        }
        cout<<"The sum :"<<sum<<endl;
    }
    
    else if(choice=='a'){
        int n,sum=0,digit,temp;
        cout<<"Enter the number :";
        cin>>n;
        temp=n;
        while(n!=0){
            digit=n%10;
            sum=sum+digit*digit*digit;
            n=n/10;
            
        }
        cout<<sum;
        if(sum==temp){
            cout<<"it is a amstrong number";
            
        }
        else{
            cout<<"it is not a amstrong number";
        }
    }
    
    else if(choice=='p'){
        int n,digit,rev=0,temp;
        cout<<"Enter the number :";
        cin>>n;
        temp=n;
        while(n!=0){
            digit=n%10;
            rev=rev*10+digit;
            n=n/10;
        }
        cout<<rev<<endl;
        if(rev==temp){
            cout<<"it is a palindrome number";
            
        }
        else{
            cout<<"it is not a palindrom number";
        }
    }
    else{
        cout<<"Invalid input";
    }
    
    return 0;
}
