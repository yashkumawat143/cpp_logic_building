14. Write a program to check whether a number is prime or not using a for loop.

#include<iostream>
using namespace std;
int main(){
    int n,i,count=0;
    cout<<"Enter the number :";
    cin>>n;
    for(i=1;i<=n;i++){
        if(n%i==0){
            count++;
        }
        
    }
    cout<<count<<endl;
    if(count==2){
        cout<<"it is a prime number";
    }
    else{
        cout<<"it is not a prime number";
    }
    
    return 0;
}
