1. Write a program to find the sum of squares of first n natural numbers using a for loop.

#include<iostream>
using namespace std;
int main(){
    int n,i,sum=0;
    cout<<"enter the number=";
    cin>>n;
    for(i=1;i<=n;i++){
        sum=sum+(i*i);
    }
    
    cout<<"sum of square="<<sum<<endl;
    
    return 0;
}
